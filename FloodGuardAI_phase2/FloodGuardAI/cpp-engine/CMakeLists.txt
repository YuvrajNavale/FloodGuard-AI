cmake_minimum_required(VERSION 3.16)
project(floodguard_engine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra)
endif()

add_library(fgcore STATIC src/terrain.cpp src/simulator.cpp src/io.cpp
    src/infrastructure.cpp src/road_graph.cpp src/infra_io.cpp src/impact.cpp src/impact_io.cpp)
target_include_directories(fgcore PUBLIC include)

add_executable(floodguard_engine src/main.cpp)
target_link_libraries(floodguard_engine PRIVATE fgcore)

enable_testing()
add_executable(test_engine tests/test_engine.cpp)
target_link_libraries(test_engine PRIVATE fgcore)
target_compile_definitions(test_engine PRIVATE SCENARIO_DIR="${CMAKE_SOURCE_DIR}/../scenarios")
add_test(NAME engine_unit_tests COMMAND test_engine)
add_executable(test_infra tests/test_infra.cpp)
target_link_libraries(test_infra PRIVATE fgcore)
target_compile_definitions(test_infra PRIVATE SCENARIO_DIR="${CMAKE_SOURCE_DIR}/../scenarios")
add_test(NAME infrastructure_tests COMMAND test_infra)
add_executable(test_impact tests/test_impact.cpp)
target_link_libraries(test_impact PRIVATE fgcore)
target_compile_definitions(test_impact PRIVATE SCENARIO_DIR="${CMAKE_SOURCE_DIR}/../scenarios")
add_test(NAME impact_tests COMMAND test_impact)
